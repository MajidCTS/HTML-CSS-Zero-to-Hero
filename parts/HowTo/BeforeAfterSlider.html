// -------- گرفتن المان‌های مورد نیاز --------
const slider = document.getElementById('slider');          // ظرف اصلی
const imageAfter = document.getElementById('imageAfter');  // تصویر رویی که برش می‌خورد
const handle = document.getElementById('sliderHandle');    // دسته اسلایدر


let isDragging = false; // آیا کاربر در حال کشیدن است؟


// -------- تابع اصلی: برش تصویر و جابجایی دسته بر اساس موقعیت ماوس --------
function updateSlider(clientX) {
    // ۱. موقعیت مستطیل ظرف را بدست می‌آوریم (offset left, width)
    const rect = slider.getBoundingClientRect();


    // ۲. محاسبه موقعیت ماوس نسبت به لبه چپ ظرف (حداقل ۰ و حداکثر عرض ظرف)
    let x = clientX - rect.left;
    x = Math.max(0, Math.min(x, rect.width)); // محدود کردن به داخل ظرف


    // ۳. تبدیل به درصد (۰ تا ۱۰۰)
    const percent = (x / rect.width) * 100;


    // ۴. اعمال به تصویر رویی (برش از راست)
    imageAfter.style.clipPath = `inset(0 ${100 - percent}% 0 0)`;


    // ۵. جابجایی دسته به همان موقعیت
    handle.style.left = `${percent}%`;
}


// -------- رویدادهای ماوس --------
// وقتی کاربر روی دسته کلیک می‌کند، حالت کشیدن فعال می‌شود
handle.addEventListener('mousedown', (e) => {
    isDragging = true;
    handle.style.cursor = 'grabbing'; // تغییر شکل ماوس
    e.preventDefault(); // جلوگیری از انتخاب متن
});


// وقتی ماوس روی صفحه حرکت می‌کند (اگر در حالت کشیدن باشیم)
document.addEventListener('mousemove', (e) => {
    if (!isDragging) return; // اگر کشیدن فعال نیست، کاری نکن
    updateSlider(e.clientX);
});


// وقتی دکمه ماوس رها می‌شود، حالت کشیدن غیرفعال می‌شود
document.addEventListener('mouseup', () => {
    if (isDragging) {
        isDragging = false;
        handle.style.cursor = 'ew-resize'; // برگشت به حالت عادی
    }
});


// -------- پشتیبانی از دستگاه‌های لمسی (موبایل و تبلت) --------
// وقتی کاربر با انگشت روی دسته لمس می‌کند
handle.addEventListener('touchstart', (e) => {
    isDragging = true;
    e.preventDefault(); // جلوگیری از اسکرول صفحه
});


// وقتی انگشت روی صفحه حرکت می‌کند
document.addEventListener('touchmove', (e) => {
    if (!isDragging) return;
    const touch = e.touches[0];
    updateSlider(touch.clientX);
    e.preventDefault(); // جلوگیری از اسکرول هنگام کشیدن
});


// وقتی انگشت از روی صفحه برداشته می‌شود
document.addEventListener('touchend', () => {
    isDragging = false;
});


// (اختیاری) اگر کاربر روی ظرف کلیک کند، دسته به همان نقطه برود
slider.addEventListener('click', (e) => {
    // فقط اگر روی دسته کلیک نکرده باشد (تا از تداخل جلوگیری شود)
    if (e.target === handle || e.target === slider) {
        updateSlider(e.clientX);
    }
});
